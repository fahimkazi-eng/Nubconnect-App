-- Example Row Level Security Guardrail:
-- Club representatives can manage their club description and events,
-- but CANNOT elevate verification_status to 'officially_verified':
CREATE POLICY "Club reps cannot self-verify" ON organizations FOR UPDATE
USING (public.is_club_rep(id) OR public.is_university_admin(university_id))
WITH CHECK (
  public.is_university_admin(university_id)
  OR verification_status = (SELECT o.verification_status FROM organizations o WHERE o.id = organizations.id)
);
